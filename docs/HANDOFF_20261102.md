nice -n 10 cargo test --release -j 12 --bin panzerrechnersystem -- <filter>

KESSEL_LOOK=deltap-tinted KESSEL_CAPTURES_DIR=/home/alex/panzer/captures KESSEL_CAPTURE_4K_JPG=1
